# OMEGL WhatsApp Bot Cron Jobs
# Server runs in UTC, Nigerian time is UTC+1 (WAT)

# Daily opening message - 7:00 AM Nigerian time (06:00 UTC)
0 6 * * * /home/ubuntu/whatsapp-bot/cron-jobs.sh daily-opening >> /home/ubuntu/whatsapp-bot/logs/cron.log 2>&1

# Pending approval reminder - every 2 hours during work hours (8 AM - 8 PM Nigerian = 7-19 UTC)
0 7,9,11,13,15,17,19 * * * /home/ubuntu/whatsapp-bot/cron-jobs.sh pending-reminder >> /home/ubuntu/whatsapp-bot/logs/cron.log 2>&1

# End of day reminder - 11:30 PM Nigerian time (22:30 UTC)
30 22 * * * /home/ubuntu/whatsapp-bot/cron-jobs.sh daily-closing >> /home/ubuntu/whatsapp-bot/logs/cron.log 2>&1

# Daily summary - 12:00 AM Nigerian time (23:00 UTC)
0 23 * * * /home/ubuntu/whatsapp-bot/cron-jobs.sh daily-summary >> /home/ubuntu/whatsapp-bot/logs/cron.log 2>&1

# Low inventory alert - every 2 hours during work hours (8 AM - 6 PM Nigerian = 7-17 UTC)
0 7,9,11,13,15,17 * * * /home/ubuntu/whatsapp-bot/cron-jobs.sh low-inventory-alert >> /home/ubuntu/whatsapp-bot/logs/cron.log 2>&1

# Backup every 30 minutes
*/30 * * * * /home/ubuntu/backup/backup.sh >> /home/ubuntu/backups/backup.log 2>&1
